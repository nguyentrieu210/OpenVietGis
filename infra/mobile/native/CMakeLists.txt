cmake_minimum_required(VERSION 3.22)
project(ovg_native_qualification LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(GDAL CONFIG REQUIRED)
find_package(PROJ CONFIG REQUIRED)
find_package(GEOS CONFIG REQUIRED)
find_package(unofficial-sqlite3 CONFIG REQUIRED)
add_executable(ovg_native_probe probe.cpp)
target_link_libraries(ovg_native_probe PRIVATE GDAL::GDAL PROJ::proj GEOS::geos_c unofficial::sqlite3::sqlite3)
if(ANDROID)
  target_link_options(ovg_native_probe PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
endif()
